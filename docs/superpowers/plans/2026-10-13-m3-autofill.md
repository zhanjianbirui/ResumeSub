# ResumeSub M3 网页自动填写 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在用户授权过的招聘网站上，页面打开后自动扫描表单，用规则词典匹配档案字段并填好。支持文本框、原生下拉框、单选组和简历文件上传。填好的字段标绿，没填的标灰，右下角浮层显示统计。数据上锁时浮层提示用户去解锁，解锁后自动继续填写。

**Architecture:**
- Content script 只在已授权的站点上运行。它由 background 通过 `chrome.scripting.registerContentScripts` 动态注册，授权的站点列表以 `chrome.permissions` 为准。
- Content script 只做三件事：扫描表单得到 `FieldDescriptor[]`、填写、标注。它不接触完整档案。
- 匹配和取值全部在 background 中完成：content 把字段描述发给 background（`autofill.plan`），background 只返回这一页要填的 `{signature, value | file}`。
- 路由层对 content script 只开放 `autofill.plan` 一个接口，并且会再检查发送方页面的 origin 是否在已授权范围内。

**Tech Stack:** 沿用 M1/M2 的技术栈（WXT MV3、TypeScript、Preact、valibot）。content script 用原生 DOM 实现，不引入 Preact。

**Spec:** `docs/superpowers/specs/2026-10-13-resume-autofill-extension-design.md`（§3、§6、§8；本计划对应 §11 的 M3）。以下内容不在本计划内：自定义下拉框、日期控件、可重复区块、Workday 和北森的专用适配器（M4），大模型兜底匹配和映射记忆（M5），开放题（M6）。

## Global Constraints

- 权限：`permissions` 为 `["storage","alarms","scripting","activeTab"]`，`optional_host_permissions` 为 `["https://*/*","http://*/*"]`。不申请 `<all_urls>`，也不把任何站点写进 manifest 的 `content_scripts`
- content script 不能 import valibot、preact，也不能 import 任何 background 模块；构建后的 `content-scripts/autofill.js` 必须小于 50KB
- content script 发给 background 的只有字段描述（标签、name、id、placeholder、autocomplete、所在区块标题、选项），不包含页面上用户已经输入的值
- background 只回传本页需要填的值；完整档案不离开 background
- 不自动点击"下一步"或"提交"；已经有值的字段不覆盖；开放题（textarea 加上问句式标签）不填写
- 浮层放在 closed Shadow DOM 中，用 `textContent` 和 `createElement` 构建，不使用 `innerHTML`；页面里不出现任何密码输入框
- 单个文件不超过 400 行，函数不超过 50 行；不就地修改入参
- 当前依赖版本：vite 8、vitest 5（钩子回调要写成带花括号的函数体）、TypeScript 7（开启了 noUncheckedIndexedAccess）、preact 11、wxt 0.21
- 测试数据只使用虚构信息；覆盖率 ≥ 80%（`entrypoints/*.content.ts`、`background.ts`、`main.tsx` 不统计）
- UI 文案使用中文；commit 格式为 `<type>: <description>`

```ts
export type FieldKind = (typeof FIELD_KINDS)[number];

export interface FieldOption {
  value: string;
  label: string;
}

export interface FieldDescriptor {
  signature: string;
  kind: FieldKind;
  label: string;
  name: string;
  id: string;
  placeholder: string;
  autocomplete: string;
  section: string;
  options?: FieldOption[];
}

export interface ResumeFileLike {
  name: string;
  mime: string;
  dataBase64: string;
}

export interface FillInstruction {
  signature: string;
  value?: string;
  file?: ResumeFileLike;
}

export type PlanResponse = { status: 'locked' } | { status: 'ready'; fills: FillInstruction[] };
```

```ts
import * as v from 'valibot';
import { FIELD_KINDS } from './autofill-types';

export const MAX_FIELDS = 300;

const text = (max: number) => v.pipe(v.string(), v.maxLength(max));

export const FieldDescriptorSchema = v.object({
  signature: text(64),
  kind: v.picklist(FIELD_KINDS),
  label: text(200),
  name: text(200),
  id: text(200),
  placeholder: text(200),
  autocomplete: text(100),
  section: text(200),
  options: v.optional(v.pipe(v.array(v.object({ value: text(200), label: text(200) })), v.maxLength(500))),
});

export const AutofillPlanSchema = v.object({
  type: v.literal('autofill.plan'),
  url: text(2048),
  fields: v.pipe(v.array(FieldDescriptorSchema), v.maxLength(MAX_FIELDS)),
});
```

```ts
  if (!text.trim()) return undefined;
  const hit = GLOBAL_RULES.find(([, re]) => re.test(text));
  if (hit) return hit[0];
  const group = sectionGroupOf(desc.section) ?? sectionGroupOf(desc.label);
  const sectionHit = SECTION_RULES.find(([, re]) => re.test(text));
  if (group && sectionHit && SECTION_KEYS[group].includes(sectionHit[0])) {
    return `${group}.${sectionHit[0]}` as FieldPath;
  }
  return undefined;
}

function autocompletePath(autocomplete: string): FieldPath | undefined {
  const token = autocomplete.trim().toLowerCase().split(/\s+/).pop() ?? '';
  return AUTOCOMPLETE[token];
}

export function matchField(desc: FieldDescriptor, customKeys: readonly string[]): FieldPath | null {
  const secondary = `${desc.placeholder} ${desc.name} ${desc.id}`;
  if (desc.kind === 'file') return RESUME_FILE.test(`${desc.label} ${secondary}`) ? 'resume.file' : null;
  if (SKIP.test(`${desc.label} ${secondary}`)) return null;
  const labelKey = normKey(desc.label);
  const custom = labelKey ? customKeys.find((key) => normKey(key) === labelKey) : undefined;
  if (custom) return `custom:${custom}`;
  const auto = autocompletePath(desc.autocomplete);
  if (auto) return auto;
  const primary = matchText(cleanLabel(desc.label), desc);
  if (primary !== undefined) return primary;
  return matchText(secondary, desc) ?? null;
}
```

```ts
  if (english && basic.nameEn?.trim()) {

  const surname = COMPOUND_SURNAMES.find((s) => name.startsWith(s) && name.length > s.length) ?? name.slice(0, 1);

function dateStyle(placeholder: string): (y: string, m: string) => string {
  if (/年/.test(placeholder)) return (y, m) => `${y}年${m}月`;
  if (/\//.test(placeholder)) return (y, m) => `${y}/${m}`;
  if (/\d\.\d|\.mm|\./i.test(placeholder)) return (y, m) => `${y}.${m}`;
  return (y, m) => `${y}-${m}`;
}

export function formatDate(value: string, field: FieldDescriptor, english: boolean): string | null {
  if (!value) return null;
  if (value === 'present') return field.kind === 'month' || field.kind === 'date' ? null : english ? 'Present' : '至今';
  if (field.kind === 'month') return value.slice(0, 7);
  if (field.kind === 'date') return value.length === 7 ? `${value}-01` : value;
  const [y = '', m = ''] = value.split('-');

}

  if (path === 'preferences.availableFrom') {

  if (path === 'skills.all') {
    const { languages, frameworks, tools } = profile.skills;
    return nonEmpty(joinList([...languages, ...frameworks, ...tools], ctx.english));

  return null;
```

```ts
};

const PLACEHOLDER = /^(?:请选择|select|choose|--|-|—)/i;
const keyOf = (text: string) => text.toLowerCase().replace(/[\s'’.\-_()（）]/g, '');
const MIN_LOOSE = 2;

export function synonymsFor(path: FieldPath, raw: string): string[] {
  if (path === 'basic.gender' || path === 'education.degree') return [...(SYNONYMS[raw] ?? [raw])];
  return [raw];
}

/** 先精确匹配（选项文字或 value），再做包含式模糊匹配；少于 2 个字符的词只做精确匹配。 */
export function chooseOption(options: readonly FieldOption[], wanted: readonly string[]): string | null {
  const candidates = options.filter((o) => o.value !== '' && !PLACEHOLDER.test(o.label.trim()));
  const keys = wanted.map(keyOf).filter(Boolean);
  const exact = candidates.find((o) => keys.includes(keyOf(o.label)) || keys.includes(keyOf(o.value)));
  if (exact) return exact.value;
  const loose = candidates.find((o) => {
    const label = keyOf(o.label);
    return label.length >= MIN_LOOSE
      && keys.some((k) => k.length >= MIN_LOOSE && (label.includes(k) || k.includes(label)));
  });
  return loose?.value ?? null;
}
```

```ts
import type { FieldDescriptor, FillInstruction } from '@/shared/autofill-types';
import type { VaultData } from '@/shared/schema/vault-data';
import { matchField } from './match';
import { chooseOption, synonymsFor } from './options';
import type { FieldPath } from './rules';
import { displayValue, resolveValue } from './values';

const CJK = /[㐀-鿿]/;

function buildFill(field: FieldDescriptor, path: FieldPath, index: number, data: VaultData): FillInstruction | null {
  if (path === 'resume.file') return data.resumeFile ? { signature: field.signature, file: data.resumeFile } : null;
  const english = !CJK.test(field.label) && field.label.trim() !== '';
  const raw = resolveValue(path, data.profile, { index, english, field });
  if (!raw) return null;
  if (field.kind === 'select' || field.kind === 'radio') {
    const chosen = chooseOption(field.options ?? [], synonymsFor(path, raw));
    return chosen === null ? null : { signature: field.signature, value: chosen };
  }
  return { signature: field.signature, value: displayValue(path, raw, english) };
}

/** 同一个 path 第 k 次出现时取第 k 条记录，用来处理多段教育或实习区块。 */
export function planFills(fields: readonly FieldDescriptor[], data: VaultData): FillInstruction[] {
  const customKeys = Object.keys(data.profile.custom);
  const counters = new Map<FieldPath, number>();
  const fills: FillInstruction[] = [];
  for (const field of fields) {
    const path = matchField(field, customKeys);
    if (!path) continue;
    const index = counters.get(path) ?? 0;
    counters.set(path, index + 1);
    const fill = buildFill(field, path, index, data);
    if (fill) fills.push(fill);
  }
  return fills;
}
```

```ts
export interface PendingTabs {
  add(tabId: number): void;
  flush(): Promise<void>;
}

/** 记录因档案上锁而没能完成填写的标签页，解锁后通知它们重试。标签页已关闭导致的发送失败是正常情况，忽略即可。 */
export function createPendingTabs(notify: (tabId: number) => Promise<unknown>): PendingTabs {
  let waiting: ReadonlySet<number> = new Set();
  return {
    add(tabId) {
      waiting = new Set([...waiting, tabId]);
    },
    async flush() {
      const tabs = [...waiting];
      waiting = new Set();
      await Promise.all(tabs.map((tabId) => notify(tabId).catch(() => undefined)));
    },
  };
}
```

```ts
export const AUTOFILL_SCRIPT_ID = 'resumesub-autofill';
export const AUTOFILL_SCRIPT_FILE = 'content-scripts/autofill.js';

interface PermissionsLike {
  getAll(): Promise<{ origins?: string[] }>;
  contains(permissions: { origins: string[] }): Promise<boolean>;
}

interface ContentScriptSpec {
  id: string;
  js: string[];
  matches: string[];
  runAt: 'document_idle';
  allFrames: boolean;
  persistAcrossSessions: boolean;
}

interface ScriptingLike {
  getRegisteredContentScripts(filter: { ids: string[] }): Promise<{ id: string }[]>;
  registerContentScripts(scripts: ContentScriptSpec[]): Promise<void>;
  updateContentScripts(scripts: ContentScriptSpec[]): Promise<void>;
  unregisterContentScripts(filter: { ids: string[] }): Promise<void>;
  executeScript(injection: { target: { tabId: number; allFrames: boolean }; files: string[] }): Promise<unknown>;
}

interface TabsLike {
  query(query: { url: string[] }): Promise<{ id?: number }[]>;
}

const SITE_PATTERN = /^https?:\/\//;

export async function grantedSites(permissions: PermissionsLike): Promise<string[]> {
  return ((await permissions.getAll()).origins ?? []).filter((o) => SITE_PATTERN.test(o));
}

/** 授权的站点列表以 chrome.permissions 为准；每次权限变化后同步 content script 的注册。 */
export async function syncAutofillScripts(scripting: ScriptingLike, permissions: PermissionsLike): Promise<void> {
  const matches = await grantedSites(permissions);
  const registered = await scripting.getRegisteredContentScripts({ ids: [AUTOFILL_SCRIPT_ID] });
  if (matches.length === 0) {
    if (registered.length > 0) await scripting.unregisterContentScripts({ ids: [AUTOFILL_SCRIPT_ID] });
    return;
  }
  const spec: ContentScriptSpec = {
    id: AUTOFILL_SCRIPT_ID, js: [AUTOFILL_SCRIPT_FILE], matches, runAt: 'document_idle', allFrames: true, persistAcrossSessions: true,
  };
  if (registered.length > 0) await scripting.updateContentScripts([spec]);
  else await scripting.registerContentScripts([spec]);
}

/** 刚授权的站点已经打开的页面不会自动注入脚本，这里补一次注入。 */
export async function injectIntoTabs(tabs: TabsLike, scripting: ScriptingLike, patterns: string[]): Promise<void> {
  const open = await tabs.query({ url: patterns });
  await Promise.all(open.map(async ({ id }) => {
    if (id === undefined) return;
    try {
      await scripting.executeScript({ target: { tabId: id, allFrames: true }, files: [AUTOFILL_SCRIPT_FILE] });
    } catch (error) {
      console.warn('[ResumeSub] inject into tab failed', id, error);
    }
  }));
}

export function createOriginCheck(permissions: PermissionsLike): (url: string) => Promise<boolean> {
  return async (url) => {
    try {
      const { origin, protocol } = new URL(url);
      if (protocol !== 'https:' && protocol !== 'http:') return false;
      return await permissions.contains({ origins: [`${origin}/*`] });
    } catch {
      return false;
    }
  };
}
```

```ts
export function fnv1a(text: string): string {
  let hash = 0x811c9dc5;
  for (let i = 0; i < text.length; i++) {
    hash ^= text.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193);
  }
  return (hash >>> 0).toString(16).padStart(8, '0');
}
```

```ts
const CONTROLS = 'input, select, textarea, button, option';
const MAX_LABEL = 60;
const HEADING = ':scope > h1, :scope > h2, :scope > h3, :scope > h4, :scope > h5, :scope > h6, :scope > legend, :scope > [role="heading"], :scope > [class*="title"]';
const MAX_SECTION_DEPTH = 8;

const clean = (text: string | null | undefined) => (text ?? '').replace(/\s+/g, ' ').trim();

/** 祖先节点去掉所有表单控件后剩下的文字（用来处理"span + input"这种没有 label 的结构）。 */
function textOutsideControls(ancestor: Element): string {
  const copy = ancestor.cloneNode(true) as Element;
  copy.querySelectorAll(CONTROLS).forEach((node) => node.remove());
  return clean(copy.textContent);
}

function controlCount(ancestor: Element): number {
  return ancestor.querySelectorAll('input:not([type="hidden"]), select, textarea').length;
}

function nearbyText(el: HTMLElement): string {
  let node: Element | null = el.parentElement;
  for (let depth = 0; node && depth < 3; depth++, node = node.parentElement) {
    if (controlCount(node) > 1) break;
    const text = textOutsideControls(node);
    if (text && text.length <= MAX_LABEL) return text;
  }
  const prev = el.previousElementSibling;
  return prev && controlCount(prev) === 0 ? clean(prev.textContent).slice(0, MAX_LABEL) : '';
}

export function resolveLabel(el: HTMLElement): string {
  const labels = (el as HTMLInputElement).labels;
  const direct = labels && labels.length > 0 ? textOutsideControls(labels[0] as Element) : '';
  if (direct) return direct;
  const labelledBy = el.getAttribute('aria-labelledby');
  if (labelledBy) {
    const text = clean(labelledBy.split(/\s+/).map((id) => el.ownerDocument.getElementById(id)?.textContent).join(' '));
    if (text) return text;
  }
  const aria = clean(el.getAttribute('aria-label'));
  if (aria) return aria;
  const near = nearbyText(el);
  if (near) return near;
  return clean(el.getAttribute('placeholder'));
}

export function sectionOf(el: HTMLElement): string {
  let node: Element | null = el.parentElement;
  for (let depth = 0; node && depth < MAX_SECTION_DEPTH; depth++, node = node.parentElement) {
    const heading = node.querySelector(HEADING);
    if (heading && !heading.contains(el)) {
      const text = clean(heading.textContent);
      if (text && text.length <= MAX_LABEL) return text;
    }
  }
  return '';
}
```

```ts
import type { FieldDescriptor, FieldKind, FieldOption } from '@/shared/autofill-types';
import { resolveLabel, sectionOf } from './label';
import { fnv1a } from './signature';

export interface ScannedField {
  descriptor: FieldDescriptor;
  element: HTMLElement;
}

const SKIP_TYPES = new Set(['hidden', 'submit', 'button', 'reset', 'image', 'password', 'checkbox', 'search', 'color', 'range']);
const INPUT_KINDS: Readonly<Record<string, FieldKind>> = {
  email: 'email', tel: 'tel', url: 'url', number: 'number', date: 'date', month: 'month', file: 'file', radio: 'radio',
};
const OPEN_QUESTION = /为什么|请描述|请简述|请介绍|谈谈|自我评价|自我介绍|[?？]|\bwhy\b|describe|tell us|cover letter/i;
const MAX_TEXT = 200;

function kindOf(el: HTMLElement): FieldKind | null {
  if (el instanceof HTMLSelectElement) return 'select';
  if (el instanceof HTMLTextAreaElement) return 'textarea';
  const type = (el as HTMLInputElement).type.toLowerCase();
  if (SKIP_TYPES.has(type)) return null;
  return INPUT_KINDS[type] ?? 'text';
}

function isHidden(el: HTMLElement): boolean {
  if (el.closest('[hidden], [aria-hidden="true"]')) return true;
  const view = el.ownerDocument.defaultView;
  for (let node: HTMLElement | null = el; node; node = node.parentElement) {
    const style = view?.getComputedStyle(node);
    if (style && (style.display === 'none' || style.visibility === 'hidden')) return true;
  }
  return false;
}

function isUsable(el: HTMLElement, kind: FieldKind): boolean {
  const control = el as HTMLInputElement;
  if (control.disabled || control.readOnly) return false;
  return kind === 'file' || !isHidden(el);
}

const cut = (text: string) => text.slice(0, MAX_TEXT);

function radioOptions(group: readonly HTMLInputElement[]): FieldOption[] {
  return group.map((radio) => ({ value: cut(radio.value), label: cut(resolveLabel(radio)) }));
}

function radioGroupLabel(group: readonly HTMLInputElement[]): string {
  const first = group[0];
  let container: HTMLElement | null = first?.parentElement ?? null;
  while (container && !group.every((r) => container?.contains(r))) container = container.parentElement;
  const holder = container?.parentElement && container.tagName === 'LABEL' ? container.parentElement : container;
  if (!holder) return '';
  const copy = holder.cloneNode(true) as HTMLElement;
  copy.querySelectorAll('label, input').forEach((n) => n.remove());
  return (copy.textContent ?? '').replace(/\s+/g, ' ').trim();
}

function describe(el: HTMLElement, kind: FieldKind, label: string, options?: FieldOption[]): Omit<FieldDescriptor, 'signature'> {
  const input = el as HTMLInputElement;
  return {
    kind, label: cut(label), name: cut(input.name ?? ''), id: cut(el.id), placeholder: cut(input.placeholder ?? ''),
    autocomplete: cut(el.getAttribute('autocomplete') ?? ''), section: cut(sectionOf(el)), ...(options ? { options } : {}),
  };
}

export function scanFields(root: ParentNode): ScannedField[] {
  const seenRadioGroups = new Set<string>();
  const counts = new Map<string, number>();
  const result: ScannedField[] = [];
  for (const el of Array.from(root.querySelectorAll<HTMLElement>('input, select, textarea'))) {
    const kind = kindOf(el);
    if (!kind || !isUsable(el, kind)) continue;
    let draft: Omit<FieldDescriptor, 'signature'>;
    if (kind === 'radio') {
      const name = (el as HTMLInputElement).name;
      if (!name || seenRadioGroups.has(name)) continue;
      seenRadioGroups.add(name);
      const group = Array.from(root.querySelectorAll<HTMLInputElement>(`input[type="radio"][name="${CSS.escape(name)}"]`));
      draft = describe(el, 'radio', radioGroupLabel(group), radioOptions(group));
    } else {
      const label = resolveLabel(el);
      if (kind === 'textarea' && OPEN_QUESTION.test(label)) continue;
      const options = el instanceof HTMLSelectElement
        ? Array.from(el.options).map((o) => ({ value: cut(o.value), label: cut(o.text.trim()) }))
        : undefined;
      draft = describe(el, kind, label, options);
    }
    const base = fnv1a(`${draft.label}|${draft.name}|${draft.id}|${draft.section}`);
    const occurrence = counts.get(base) ?? 0;
    counts.set(base, occurrence + 1);
    result.push({ descriptor: { ...draft, signature: `${base}-${occurrence}` }, element: el });
  }
  return result;
}
```

```ts
import type { FillInstruction, ResumeFileLike } from '@/shared/autofill-types';
import type { ScannedField } from './scan';

export type FillOutcome = 'filled' | 'skipped' | 'failed';

interface FillDeps {
  makeDataTransfer?: () => DataTransfer;
}

const BASE64 = /^[A-Za-z0-9+/]*={0,2}$/;

function fire(el: HTMLElement, ...types: string[]) {
  for (const type of types) el.dispatchEvent(new Event(type, { bubbles: true }));
}

/** 必须用原型上的 setter，React、Vue 才能察觉值的变化。 */
function setNativeValue(el: HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement, value: string) {
  const proto = Object.getPrototypeOf(el) as object;
  const setter = Object.getOwnPropertyDescriptor(proto, 'value')?.set;
  if (setter) setter.call(el, value);
  else el.value = value;
}

function fillText(el: HTMLInputElement | HTMLTextAreaElement, value: string): FillOutcome {
  if (el.value.trim() !== '') return 'skipped';
  setNativeValue(el, value);
  fire(el, 'input', 'change');
  return 'filled';
}

function fillSelect(el: HTMLSelectElement, value: string): FillOutcome {
  if (el.value !== '' && el.selectedIndex > 0) return 'skipped';
  if (!Array.from(el.options).some((o) => o.value === value)) return 'failed';
  setNativeValue(el, value);
  fire(el, 'input', 'change');
  return 'filled';
}

function fillRadio(el: HTMLInputElement, value: string, root: ParentNode): FillOutcome {
  const group = Array.from(root.querySelectorAll<HTMLInputElement>('input[type="radio"]')).filter((r) => r.name === el.name);
  if (group.some((r) => r.checked)) return 'skipped';
  const target = group.find((r) => r.value === value);
  if (!target) return 'failed';
  target.click();
  return 'filled';
}

function decode(file: ResumeFileLike): File | null {
  if (!BASE64.test(file.dataBase64)) return null;
  try {
    const binary = atob(file.dataBase64);
    const bytes = Uint8Array.from(binary, (c) => c.charCodeAt(0));
    return new File([bytes], file.name, { type: file.mime });
  } catch {
    return null;
  }
}

function fillFile(el: HTMLInputElement, file: ResumeFileLike, deps: FillDeps): FillOutcome {
  if (el.files && el.files.length > 0) return 'skipped';
  const decoded = decode(file);
  if (!decoded) return 'failed';
  const transfer = (deps.makeDataTransfer ?? (() => new DataTransfer()))();
  transfer.items.add(decoded);
  el.files = transfer.files;
  fire(el, 'input', 'change');
  return 'filled';
}

export function applyFill(field: ScannedField, fill: FillInstruction, root: ParentNode, deps: FillDeps = {}): FillOutcome {
  const el = field.element;
  const kind = field.descriptor.kind;
  try {
    if (kind === 'file') return fill.file ? fillFile(el as HTMLInputElement, fill.file, deps) : 'failed';
    if (fill.value === undefined) return 'failed';
    if (kind === 'radio') return fillRadio(el as HTMLInputElement, fill.value, root);
    if (el instanceof HTMLSelectElement) return fillSelect(el, fill.value);
    return fillText(el as HTMLInputElement, fill.value);
  } catch (error) {
    console.warn('[ResumeSub] fill failed', field.descriptor.label, error);
    return 'failed';
  }
}
```

```ts
export type OverlayState =
  | { kind: 'locked' }
  | { kind: 'result'; filled: number; unfilled: number }
  | { kind: 'error'; message: string };

export interface Overlay {
  show(state: OverlayState): void;
  text(): string;
  destroy(): void;
}

const HOST_ID = 'resumesub-overlay';
const CSS = `
  .box { position: fixed; right: 16px; bottom: 16px; z-index: 2147483647; font: 13px/1.5 system-ui, -apple-system, 'PingFang SC', sans-serif;
    background: #fff; color: #1f2328; border: 1px solid #d0d7de; border-radius: 8px; padding: 8px 12px; box-shadow: 0 4px 12px rgba(0,0,0,.12); cursor: pointer; max-width: 320px; }
  .brand { font-weight: 600; margin-right: 6px; }
  .warn { color: #9a6700; } .error { color: #cf222e; }
`;

function describeState(state: OverlayState): { text: string; tone: string } {
  if (state.kind === 'locked') return { text: '🔒 档案已上锁：点击浏览器工具栏中的 ResumeSub 图标解锁后自动填写', tone: 'warn' };
  if (state.kind === 'error') return { text: `自动填写失败：${state.message}`, tone: 'error' };
  return { text: `已填 ${state.filled} · 未填 ${state.unfilled}（点击跳到未填字段）`, tone: '' };
}

export function createOverlay(doc: Document, onNext: () => void): Overlay {
  const host = doc.createElement('div');
  host.id = HOST_ID;
  const shadow = host.attachShadow({ mode: 'closed' });
  const style = doc.createElement('style');
  style.textContent = CSS;
  const box = doc.createElement('div');
  box.className = 'box';
  box.addEventListener('click', onNext);
  shadow.append(style, box);
  return {
    show(state) {
      const { text, tone } = describeState(state);
      const brand = doc.createElement('span');
      brand.className = 'brand';
      brand.textContent = 'ResumeSub';
      const body = doc.createElement('span');
      body.className = tone;
      body.textContent = text;
      box.replaceChildren(brand, body);
      if (!host.isConnected) doc.documentElement.appendChild(host);
    },
    text: () => box.textContent ?? '',
    destroy: () => host.remove(),
  };
}
```

```ts
import type { FieldDescriptor, FillInstruction, PlanResponse } from '@/shared/autofill-types';
import { applyFill } from './fill';
import { markField } from './mark';
import type { Overlay } from './overlay';
import { type ScannedField, scanFields } from './scan';

export interface SessionDeps {
  doc: Document;
  requestPlan(fields: FieldDescriptor[]): Promise<PlanResponse>;
  overlay: Overlay;
}

export function waitForStable(root: Node, quietMs = 500, maxMs = 5000): Promise<void> {
  return new Promise((resolve) => {
    let quietTimer: ReturnType<typeof setTimeout>;
    const finish = () => {
      observer.disconnect();
      clearTimeout(quietTimer);
      clearTimeout(maxTimer);
      resolve();
    };
    const observer = new MutationObserver(() => {
      clearTimeout(quietTimer);
      quietTimer = setTimeout(finish, quietMs);
    });
    observer.observe(root, { childList: true, subtree: true, attributes: true });
    quietTimer = setTimeout(finish, quietMs);
    const maxTimer = setTimeout(finish, maxMs);
  });
}

function applyAll(fields: readonly ScannedField[], fills: readonly FillInstruction[], doc: Document) {
  const bySignature = new Map(fills.map((f) => [f.signature, f]));
  let filled = 0;
  const unfilled: HTMLElement[] = [];
  for (const field of fields) {
    const fill = bySignature.get(field.descriptor.signature);
    const outcome = fill ? applyFill(field, fill, doc) : 'failed';
    if (outcome === 'filled') {
      filled += 1;
      markField(field.element, 'filled');
    } else if (outcome === 'failed') {
      unfilled.push(field.element);
      markField(field.element, 'unfilled');
    }
  }
  return { filled, unfilled };
}

/** 每次 run 只处理本次新出现的字段（按 signature 判断）；上锁时不把字段记为已处理，解锁后可以重试。 */
export function createAutofillSession(deps: SessionDeps) {
  let handled: ReadonlySet<string> = new Set();
  let filledTotal = 0;
  let unfilled: readonly HTMLElement[] = [];
  let busy = false;

  async function run(): Promise<void> {
    if (busy) return;
    busy = true;
    try {
      const fresh = scanFields(deps.doc).filter((f) => !handled.has(f.descriptor.signature));
      if (fresh.length === 0) return;
      const plan = await deps.requestPlan(fresh.map((f) => f.descriptor));
      if (plan.status === 'locked') {
        deps.overlay.show({ kind: 'locked' });
        return;
      }
      const result = applyAll(fresh, plan.fills, deps.doc);
      handled = new Set([...handled, ...fresh.map((f) => f.descriptor.signature)]);
      filledTotal += result.filled;
      unfilled = [...unfilled.filter((el) => el.isConnected), ...result.unfilled];
      deps.overlay.show({ kind: 'result', filled: filledTotal, unfilled: unfilled.length });
    } catch (error) {
      console.error('[ResumeSub] autofill failed', error);
      deps.overlay.show({ kind: 'error', message: '请刷新页面重试' });
    } finally {
      busy = false;
    }
  }

  function nextUnfilled(): void {
    const target = unfilled.find((el) => el.isConnected && el.dataset.resumesub === 'unfilled');
    target?.scrollIntoView({ block: 'center', behavior: 'smooth' });
    target?.focus();
  }

  return { run, nextUnfilled };
}
```

```ts
import { browser } from 'wxt/browser';
import { createOverlay } from '@/content/overlay';
import { createAutofillSession, waitForStable } from '@/content/run';
import type { FieldDescriptor, PlanResponse } from '@/shared/autofill-types';

const LOADED_FLAG = '__resumesubAutofillLoaded';
const RERUN_DEBOUNCE_MS = 800;
const MAX_RUNS = 20;

async function requestPlan(fields: FieldDescriptor[]): Promise<PlanResponse> {
  const res = (await browser.runtime.sendMessage({ type: 'autofill.plan', url: location.href, fields })) as
    | { ok: true; data: PlanResponse }
    | { ok: false; error: { message: string } }
    | undefined;
  if (!res) throw new Error('no response');
  if (!res.ok) throw new Error(res.error.message);
  return res.data;
}

export default defineContentScript({
  registration: 'runtime',
  async main() {
    const scope = globalThis as Record<string, unknown>;
    if (scope[LOADED_FLAG]) return;
    scope[LOADED_FLAG] = true;

    let session: ReturnType<typeof createAutofillSession> | undefined;
    const overlay = createOverlay(document, () => session?.nextUnfilled());
    session = createAutofillSession({ doc: document, requestPlan, overlay });

    await waitForStable(document.body);
    await session.run();

    let runs = 1;
    let timer: ReturnType<typeof setTimeout> | undefined;
    const observer = new MutationObserver(() => {
      clearTimeout(timer);
      timer = setTimeout(() => {
        runs += 1;
        if (runs > MAX_RUNS) observer.disconnect();
        else void session?.run();
      }, RERUN_DEBOUNCE_MS);
    });
    observer.observe(document.body, { childList: true, subtree: true });

    browser.runtime.onMessage.addListener((message: unknown) => {
      if ((message as { type?: string } | null)?.type === 'autofill.retry') void session?.run();
    });
  },
});
```

`src/shared/site-access.ts`：
```ts
import { browser } from 'wxt/browser';

export const originPattern = (origin: string) => `${origin}/*`;

export async function currentTabOrigin(): Promise<string | null> {
  const [tab] = await browser.tabs.query({ active: true, currentWindow: true });
  try {
    const url = new URL(tab?.url ?? '');
    return url.protocol === 'https:' || url.protocol === 'http:' ? url.origin : null;
  } catch {
    return null;
  }
}

export const isSiteEnabled = (origin: string) => browser.permissions.contains({ origins: [originPattern(origin)] });
export const enableSites = (patterns: string[]) => browser.permissions.request({ origins: patterns });
export const disableSites = (patterns: string[]) => browser.permissions.remove({ origins: patterns });

export async function listEnabledSites(): Promise<string[]> {
  return ((await browser.permissions.getAll()).origins ?? []).filter((o) => /^https?:\/\//.test(o));
}
```

`src/entrypoints/popup/SiteToggle.tsx`：
```tsx
import { useEffect, useState } from 'preact/hooks';
import { currentTabOrigin, disableSites, enableSites, isSiteEnabled, originPattern } from '@/shared/site-access';

type State = { kind: 'loading' } | { kind: 'unsupported' } | { kind: 'ready'; origin: string; enabled: boolean };

export function SiteToggle() {
  const [state, setState] = useState<State>({ kind: 'loading' });
  const [error, setError] = useState('');

  useEffect(() => {
    void (async () => {
      const origin = await currentTabOrigin();
      setState(origin ? { kind: 'ready', origin, enabled: await isSiteEnabled(origin) } : { kind: 'unsupported' });
    })().catch((e: unknown) => {
      console.error('[ResumeSub] read site state failed', e);
      setState({ kind: 'unsupported' });
    });
  }, []);

  if (state.kind === 'loading') return null;
  if (state.kind === 'unsupported') return <p class="hint">当前页面不支持自动填写</p>;

  // permissions.request 必须在点击事件中同步发起，否则浏览器会认为不是用户操作而拒绝
  const toggle = () => {
    setError('');
    const patterns = [originPattern(state.origin)];
    const action = state.enabled ? disableSites(patterns) : enableSites(patterns);
    action
      .then((ok) => {
        if (ok) setState({ ...state, enabled: !state.enabled });
        else setError('未获得授权');
      })
      .catch((e: unknown) => {
        console.error('[ResumeSub] toggle site failed', e);
        setError('操作失败，请重试');
      });
  };

  return (
    <section>
      <p class="hint">{state.origin}{state.enabled ? '：已启用自动填写（刷新页面后生效）' : ''}</p>
      <button type="button" class={state.enabled ? '' : 'primary'} onClick={toggle}>
        {state.enabled ? '在此网站停用' : '在此网站启用自动填写'}
      </button>
      {error && <p class="error" role="alert">{error}</p>}
    </section>
  );
}
```

```tsx
import { useEffect, useState } from 'preact/hooks';
import { disableSites, enableSites, listEnabledSites } from '@/shared/site-access';
import { PRESET_SITES } from '@/shared/sites';

export function SitesForm() {
  const [sites, setSites] = useState<string[]>([]);
  const [error, setError] = useState('');

  const refresh = () => listEnabledSites().then(setSites).catch((e: unknown) => {
    console.error('[ResumeSub] list sites failed', e);
    setError('读取授权站点失败');
  });

  useEffect(() => {
    void refresh();
  }, []);

  const run = (action: Promise<boolean>) => {
    setError('');
    action.then((ok) => (ok ? refresh() : setError('未获得授权'))).catch((e: unknown) => {
      console.error('[ResumeSub] update sites failed', e);
      setError('操作失败，请重试');
    });
  };

  return (
    <section>
      <h2>自动填写站点</h2>
      <p class="hint">只有授权过的网站才会自动填写。常见招聘系统：{[...new Set(PRESET_SITES.map((s) => s.name))].join('、')}。</p>
      <button type="button" class="primary" onClick={() => run(enableSites(PRESET_SITES.map((s) => s.pattern)))}>
        一键授权常见招聘网站
      </button>
      <ul>
        {sites.map((site) => (
          <li key={site} class="row">
            <span>{site}</span>
            <button type="button" onClick={() => run(disableSites([site]))}>移除</button>
          </li>
        ))}
      </ul>
      {error && <p class="error" role="alert">{error}</p>}
    </section>
  );
}
```
